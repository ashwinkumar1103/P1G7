fig, axes = plt.subplots(3, 2, figsize=(16, 15))

# 1. Customer churn distribution
churn_counts.plot(kind="bar", ax=axes[0, 0])
axes[0, 0].set_title("Customer Churn Distribution")
axes[0, 0].set_xlabel("Customer Status")
axes[0, 0].set_ylabel("Number of Customers")
axes[0, 0].set_xticklabels(
    ["Active Customers", "Churned Customers"],
    rotation=0
)

# 2. Churn by subscription type
churn_by_subscription.plot(
    kind="bar", stacked=True, ax=axes[0, 1]
)
axes[0, 1].set_title("Churn by Subscription Type")
axes[0, 1].set_xlabel("Subscription Type")
axes[0, 1].set_ylabel("Percentage of Customers")

# 3. Average spending
average_spending.plot(kind="bar", ax=axes[1, 0])
axes[1, 0].set_title("Average Spending by Customer Status")
axes[1, 0].set_ylabel("Average Spending")
axes[1, 0].tick_params(axis="x", rotation=0)

# 4. Average complaints
average_complaints.plot(kind="bar", ax=axes[1, 1])
axes[1, 1].set_title("Average Complaints by Customer Status")
axes[1, 1].set_ylabel("Average Complaints")
axes[1, 1].tick_params(axis="x", rotation=0)

# 5. Average login activity
average_logins.plot(kind="bar", ax=axes[2, 0])
axes[2, 0].set_title("Average Login Activity by Customer Status")
axes[2, 0].set_ylabel("Average Login Events")
axes[2, 0].tick_params(axis="x", rotation=0)

# Hide the unused sixth panel
